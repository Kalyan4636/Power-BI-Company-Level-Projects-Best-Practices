let
    Source = Csv.Document(
        File.Contents("C:\YOUR_PATH\Coffee_Cleaned.csv"),
        [Delimiter=",", Encoding=65001, QuoteStyle=QuoteStyle.Csv]
    ),
    PromotedHeaders = Table.PromoteHeaders(Source, [PromoteAllScalars=true]),
    TrimmedText = Table.TransformColumns(
        PromotedHeaders,
        {
            {"cash_type", each Text.Trim(_), type text},
            {"coffee_name", each Text.Trim(_), type text},
            {"Time_of_Day", each Text.Trim(_), type text},
            {"Weekday", each Text.Trim(_), type text},
            {"Month_name", each Text.Trim(_), type text}
        }
    ),
    ChangedTypes = Table.TransformColumnTypes(
        TrimmedText,
        {
            {"hour_of_day", Int64.Type},
            {"cash_type", type text},
            {"Spend", Currency.Type},
            {"coffee_name", type text},
            {"Time_of_Day", type text},
            {"Weekday", type text},
            {"Month_name", type text},
            {"Weekdaysort", Int64.Type},
            {"Monthsort", Int64.Type},
            {"Date", type date},
            {"Time", type text}
        },
        "en-US"
    ),
    FilterValidDates = Table.SelectRows(ChangedTypes, each [Date] <> null),
    FilterValidAmounts = Table.SelectRows(FilterValidDates, each [Spend] <> null),
    AddTransactionID = Table.AddIndexColumn(FilterValidAmounts, "TransactionID", 1, 1, Int64.Type)
in
    AddTransactionID
